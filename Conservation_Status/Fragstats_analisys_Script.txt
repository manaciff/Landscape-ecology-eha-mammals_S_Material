# PATCH ANALYSIS 
data_patch <- read.csv2(table_patch, header = TRUE, sep = ",",
                        dec = ".") # Reading the data frame
names(data_patch)
str(data_patch)

data_patch_2 <- data_patch %>%
  mutate(CLASS_TYPE = str_trim(CLASS_TYPE),
         CLASS_NAME = case_when(
           CLASS_TYPE == "cls_1"  ~ "Forest",
           CLASS_TYPE == "cls_2"  ~ "Herbaceous and Shrubby Vegetation",
           CLASS_TYPE == "cls_3"  ~ "Pasture",
           CLASS_TYPE == "cls_4"  ~ "Agriculture",
           CLASS_TYPE == "cls_5"  ~ "Water",
           CLASS_TYPE == "cls_6"  ~ "Non vegetated area",
           CLASS_TYPE == "cls_70" ~ "Roads and Railroad",
           TRUE                   ~ NA_character_
         ))
#data_patch_clean_col <- data_patch_2 %>% select(-CLASS_TYPE)

write.csv(data_patch_2, file = "Patch_Analisys_90_20.csv", row.names = TRUE)


# Statistical summary without normalization 
summary_patch <- data_patch_2 %>%
  group_by(YEAR, CLASS_NAME) %>%
  summarise(
    min_AREA  = min(AREA, na.rm = TRUE),
    max_AREA  = max(AREA, na.rm = TRUE),
    mean_AREA = mean(AREA, na.rm = TRUE),
    sd_AREA   = sd(AREA, na.rm = TRUE),
    median_AREA = median(AREA, na.rm = TRUE),
    
    min_FRAC  = min(FRAC, na.rm = TRUE),
    max_FRAC  = max(FRAC, na.rm = TRUE),
    mean_FRAC = mean(FRAC, na.rm = TRUE),
    sd_FRAC   = sd(FRAC, na.rm = TRUE),
    median_FRAC = median(FRAC, na.rm = TRUE),
    
    min_PROX  = min(PROX, na.rm = TRUE),
    max_PROX  = max(PROX, na.rm = TRUE),
    mean_PROX = mean(PROX, na.rm = TRUE),
    sd_PROX   = sd(PROX, na.rm = TRUE),
    median_PROX = median(PROX, na.rm = TRUE),
    
    min_ENN   = min(ENN, na.rm = TRUE),
    max_ENN   = max(ENN, na.rm = TRUE),
    mean_ENN  = mean(ENN, na.rm = TRUE),
    sd_ENN    = sd(ENN, na.rm = TRUE),
    median_ENN = median(ENN, na.rm = TRUE),
    .groups = "drop")

print(summary_patch)
write.csv(summary_patch, file = "Patch_summary.csv", row.names = TRUE)


# Statistical normalization using [(x - mean) / sd]
data_patch_z <- data_patch_2 %>%
  group_by(YEAR, CLASS_NAME) %>%
  mutate(across(c(AREA, FRAC, PROX, ENN),
                ~ (. - mean(., na.rm = TRUE)) / sd(., na.rm = TRUE),
                .names = "z_{.col}")) %>%
  ungroup()

#write.csv(data_patch_z, file = "Patch_Z_Analisys_90_20.csv", row.names = TRUE)

# Tests of normality 

 ## AREA    # R: no normal distribution
  qqnorm(data_patch_z$z_AREA, main = "Q-Q plot para z_AREA")
  qqline(data_patch_z$z_AREA, col = "red")

  ggplot(data_patch_z, aes(sample = z_AREA)) +
    stat_qq() +
    stat_qq_line() +
    facet_wrap(~ CLASS_NAME) +
    labs(title = "Q-Q plot for z_AREA by class")
  
   shapiro.test(sample(data_patch_z$z_AREA, 5000)) #  W = 0.056487, p-value < 2.2e-16

# ------------ USING LOG AND Z-SCORE ---------- #
  # Log and standardization  
   data_patch_z_log <- data_patch_2 %>%
     mutate(
       log_AREA = log1p(AREA),
       log_FRAC = log1p(FRAC),
       log_PROX = log1p(PROX),
       log_ENN  = log1p(ENN)
     ) %>%
     group_by(YEAR, CLASS_NAME) %>%
     mutate(
       z_log_AREA = scale(log_AREA),
       z_log_FRAC = scale(log_FRAC),
       z_log_PROX = scale(log_PROX),
       z_log_ENN  = scale(log_ENN)
     ) %>%
     ungroup()
   
   write.csv(data_patch_z_log, "Patch_Z_Log_Analisys_90_20.csv", row.names = FALSE)
   
   shapiro.test(sample(data_patch_z_log$z_log_AREA, 5000)) # W = 0.76168, p-value < 2.2e-16
   
   # Summary of data post log and standardization by median
   summary_patch_z_log <- data_patch_z_log %>%
     group_by(YEAR, CLASS_NAME) %>%
     summarise(
       med_z_log_AREA = median(z_log_AREA, na.rm = TRUE),
       iqr_z_log_AREA = IQR(z_log_AREA, na.rm = TRUE),
       
       med_z_log_FRAC = median(z_log_FRAC, na.rm = TRUE),
       iqr_z_log_FRAC = IQR(z_log_FRAC, na.rm = TRUE),
       
       med_z_log_PROX = median(z_log_PROX, na.rm = TRUE),

       iqr_z_log_ENN  = IQR(z_log_ENN, na.rm = TRUE),
       .groups = "drop"
     )
